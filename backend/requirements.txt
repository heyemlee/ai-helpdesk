# ai-helpdesk 依赖清单

# ── AI / LLM ──────────────────────────────────────────────────────────────────
# 两个 SDK 按 .env 里的 LLM_PROVIDER 二选一使用，core/openai_compat.py 把 OpenAI
# 那条路翻译成同一套 messages 接口，业务代码只写一份调用。
anthropic==1.3.0           # LLM_PROVIDER=anthropic（含兼容该协议的第三方网关）
                           # 1.x 的两处硬性变化，升级时都不会报错、只会静默变味：
                           # 1) temperature/top_p/top_k 从 messages.* 的签名里删除了，
                           #    再传是 TypeError 而不是 400。调用点一处都不传，
                           #    确定性交给结构化输出，见 core/llm_utils.py。
                           # 2) HTTP 层从 httpx 换成 httpx2，OTel 的 httpx 埋点会
                           #    看不到 SDK 的请求 —— api/main.py 顶部用
                           #    httpx2.alias_httpx() 把两者接回同一个模块。
openai==1.109.1            # LLM_PROVIDER=openai（含兼容 /chat/completions 的第三方网关）
                           # 注意 <1.55.3 的版本会给 httpx 传已被移除的 proxies 参数，
                           # 和这里锁的 httpx 0.28 撞车，建单条请求时就 TypeError
                           # 意图识别的 Embedding 一路始终走 OpenAI 兼容端点（httpx 直连），
                           # 未配置时退化为本地 n-gram 向量，三路融合链路不中断

# 升到 0.119 是为了把 Starlette 的范围放开到 0.48。

starlette==0.48.0           # 显式声明：GHSA-2c2j-9gv5-cj73 在 0.47.2 修复，
                            # 这里跟随 FastAPI 0.119 的上界取 0.48.0。
                            # 写出来是为了让"我们要的是 >=0.47.2"这件事可见——
                            # 只写 fastapi 的话，下一次有人回退 FastAPI 版本时
                            # Starlette 会跟着悄悄退回有漏洞的版本
python-multipart==0.0.32    # FastAPI 文件上传支持。0.0.12 有两个 DoS
                            # （GHSA-59g5-xgcq-4qw3 / GHSA-pp6c-gr5w-3c5g），
                            # 修复在 0.0.27
